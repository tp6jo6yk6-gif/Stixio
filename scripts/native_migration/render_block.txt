function renderActiveWorkspace() {
  if (state.activeEditor === 'refine') return renderRefineWorkspace();
  if (state.activeEditor === 'review') return renderReviewWorkspace();
  if (state.activeEditor === 'package') return renderPackageWorkspace();
  return renderLayoutWorkspace();
}

function renderLayoutWorkspace() {
  return `<aside data-workspace-column="left" class="space-y-4">${renderImportPanel()}${renderDetectionPanel()}${renderDestinationPanel()}</aside><section data-workspace-column="center" class="space-y-4">${state.sources.size ? renderSourceEditor() : renderWorkspaceEmpty('layout')}</section><aside data-workspace-column="right" class="space-y-4">${renderSourceListPanel()}${renderSelectedPanel()}</aside>`;
}

function renderRefineWorkspace() {
  return `<aside data-workspace-column="left" class="space-y-4">${renderRefinePanel()}</aside><section data-workspace-column="center" class="space-y-4">${frames().length ? renderRefineEditor() : renderWorkspaceEmpty('refine')}</section><aside data-workspace-column="right" class="space-y-4">${renderSourceListPanel()}${renderSelectedPanel()}</aside>`;
}

function renderReviewWorkspace() {
  return `<section data-workspace-column="center" class="space-y-4">${frames().length ? renderReviewBoard() : renderWorkspaceEmpty('review')}</section><aside data-workspace-column="right" class="space-y-4">${renderReviewPanel()}</aside>`;
}

function renderPackageWorkspace() {
  return `<aside data-workspace-column="left" class="space-y-4">${renderDestinationPanel()}${renderOutputPanel()}</aside><section data-workspace-column="center" class="space-y-4">${frames().length ? renderPackageBoard() : renderWorkspaceEmpty('package')}</section><aside data-workspace-column="right" class="space-y-4">${renderPackagePanel()}</aside>`;
}

function renderWorkspaceFooter(progress) {
  const index = WORKFLOW_STAGES.findIndex(stage => stage.id === state.activeEditor);
  const previous = WORKFLOW_STAGES[index - 1];
  const next = WORKFLOW_STAGES[index + 1];
  const current = progress[state.activeEditor];
  const previousButton = previous ? `<button type="button" data-workflow-direction="previous" class="rounded-xl bg-slate-100 px-4 py-2 text-xs font-black text-slate-700">← 返回：${previous.title}</button>` : '<span></span>';
  const nextButton = next ? `<button type="button" data-workflow-direction="next" data-tone="${progress[next.id].tone}" class="rounded-xl bg-slate-950 px-4 py-2 text-left text-xs font-black text-white shadow-sm"><span class="block">下一步：${next.title} →</span><span data-workflow-next-reason class="mt-0.5 block text-[10px] font-bold text-white/60">${progress[next.id].reason}</span></button>` : `<div class="rounded-xl ${statusSurfaceClass(current.tone)} px-4 py-2 text-right text-xs font-black"><span class="block">流程狀態</span><span class="mt-0.5 block text-[10px] font-bold opacity-70">${current.reason}</span></div>`;
  return `<footer data-workflow-footer class="workflow-footer rounded-[1.5rem] border border-slate-900/10 bg-white px-4 py-3 shadow-sm">${previousButton}<span class="text-center text-xs font-bold text-slate-400">自動儲存目前專案進度</span>${nextButton}</footer>`;
}

function renderWorkspaceEmpty(stageId) {
  const copy = stageId === 'layout'
    ? { title: '尚未匯入圖片', description: '加入一張或多張排版圖，開始建立貼圖裁切範圍。', action: 'file', label: '選擇圖片' }
    : { title: `尚無貼圖可${stageId === 'refine' ? '修整' : stageId === 'review' ? '檢查' : '打包'}`, description: '請先至「匯入與切割」加入圖片並建立貼圖範圍。', action: 'layout', label: '前往匯入與切割' };
  const step = WORKFLOW_STAGES.find(stage => stage.id === stageId)?.step || '01';
  return `<section data-workflow-empty="${stageId}" class="grid min-h-[420px] place-items-center rounded-[2rem] border-2 border-dashed border-slate-300 bg-white p-8 text-center shadow-sm"><div class="max-w-md"><div class="mx-auto grid h-16 w-16 place-items-center rounded-3xl bg-slate-950 text-xl font-black text-emerald-300">${step}</div><h3 class="mt-6 text-2xl font-black">${copy.title}</h3><p class="mt-3 text-sm font-bold leading-6 text-slate-500">${copy.description}</p><button type="button" data-workflow-empty-action="${copy.action}" class="mt-6 rounded-2xl bg-slate-950 px-5 py-3 text-sm font-black text-white">${copy.label}</button></div></section>`;
}

function renderMobileWorkflowNavigation() {
  const progress = getWorkflowProgress();
  return `<nav data-mobile-workflow-nav aria-label="行動版貼圖製作流程">${WORKFLOW_STAGES.map(stage => {
    const active = stage.id === state.activeEditor;
    const status = progress[stage.id];
    return `<button type="button" data-workflow-stage="${stage.id}" data-tone="${status.tone}" data-active="${active}" ${active ? 'aria-current="step"' : ''}><span>${stage.step}</span><strong>${stageShortName(stage.id)}</strong><small>${status.shortLabel}</small></button>`;
  }).join('')}</nav>`;
}

function getWorkflowProgress() {
  const sourceCount = state.sources.size;
  const frameList = frames();
  const frameCount = frameList.length;
  const frameIds = new Set(frameList.map(frame => frame.id));
  const refinedCount = [...state.refineAppliedAt.keys()].filter(id => frameIds.has(id)).length;
  const selected = exportFrames();
  const selectedCount = selected.length;
  const approvedCount = selected.filter(frame => frame.state?.reviewApproved === true).length;
  const errors = state.reviewReport?.summary?.errors || 0;
  const warnings = state.reviewReport?.summary?.warnings || 0;
  const packageReady = Boolean(state.reviewReport?.canPackage);
  const layoutDone = sourceCount > 0 && frameCount > 0;
  const refineDone = frameCount > 0 && refinedCount >= frameCount;
  const reviewDone = selectedCount > 0 && approvedCount >= selectedCount && errors === 0;
  const reviewDetail = errors ? `${errors} 錯誤` : warnings ? `${warnings} 警告` : approvedCount ? '部分核准' : '等待核准';
  return {
    layout: {
      label: sourceCount ? `${sourceCount} 張原圖 · ${frameCount} 張貼圖` : '尚未加入素材',
      shortLabel: sourceCount ? `${frameCount} 張` : '未開始',
      reason: layoutDone ? '已建立貼圖範圍' : sourceCount ? '尚未建立貼圖範圍' : '請先匯入圖片',
      percent: layoutDone ? 100 : sourceCount ? 45 : 0,
      tone: layoutDone ? 'complete' : sourceCount ? 'active' : 'idle',
      completed: layoutDone
    },
    refine: {
      label: frameCount ? `已完成 ${refinedCount}/${frameCount}` : '等待建立貼圖範圍',
      shortLabel: frameCount ? `${refinedCount}/${frameCount}` : '未開始',
      reason: frameCount ? (refineDone ? '所有貼圖已套用修圖' : `尚有 ${Math.max(0, frameCount - refinedCount)} 張未修整`) : '請先完成匯入與切割',
      percent: frameCount ? Math.round(refinedCount / frameCount * 100) : 0,
      tone: refineDone ? 'complete' : refinedCount ? 'active' : 'idle',
      completed: refineDone
    },
    review: {
      label: selectedCount ? `已核准 ${approvedCount}/${selectedCount}${errors || warnings ? ` · ${reviewDetail}` : ''}` : '尚無貼圖可檢查',
      shortLabel: selectedCount ? `${approvedCount}/${selectedCount}` : '未開始',
      reason: selectedCount ? (errors ? `仍有 ${errors} 個錯誤待修正` : warnings ? `仍有 ${warnings} 個警告待確認` : reviewDone ? '已完成檢查與核准' : `尚有 ${Math.max(0, selectedCount - approvedCount)} 張未核准`) : '請先建立可輸出的貼圖',
      percent: selectedCount ? Math.round(approvedCount / selectedCount * 100) : 0,
      tone: errors ? 'error' : warnings ? 'warning' : reviewDone ? 'complete' : approvedCount ? 'active' : 'idle',
      completed: reviewDone
    },
    package: {
      label: packageReady ? `可下載 · ${selectedCount} 張貼圖` : errors ? `${errors} 項錯誤待修正` : selectedCount ? `尚缺 ${Math.max(0, selectedCount - approvedCount)} 張核准` : '完成檢查後即可下載',
      shortLabel: packageReady ? '可下載' : '未就緒',
      reason: packageReady ? '貼圖包已符合下載條件' : errors ? `仍有 ${errors} 個錯誤待修正` : selectedCount ? `尚有 ${Math.max(0, selectedCount - approvedCount)} 張未核准` : '請先完成檢查與核准',
      percent: packageReady ? 100 : selectedCount ? Math.round(approvedCount / selectedCount * 100) : 0,
      tone: packageReady ? 'complete' : errors ? 'error' : warnings ? 'warning' : approvedCount ? 'active' : 'idle',
      completed: packageReady
    }
  };
}

function refreshWorkflowProgress() {
  const progress = getWorkflowProgress();
  WORKFLOW_STAGES.forEach(stage => {
    const status = progress[stage.id];
    const label = document.querySelector(`[data-workflow-progress-label="${stage.id}"]`);
    const reason = document.querySelector(`[data-workflow-progress-reason="${stage.id}"]`);
    const bar = document.querySelector(`[data-workflow-progress-bar="${stage.id}"]`);
    const pill = document.querySelector(`[data-workflow-status-pill="${stage.id}"]`);
    const tab = document.querySelector(`[data-workflow-stage="${stage.id}"]`);
    if (label) label.textContent = status.label;
    if (reason) reason.textContent = status.reason;
    if (bar) { bar.className = `block h-full rounded-full ${progressBarClass(status.tone)}`; bar.style.width = `${status.percent}%`; }
    if (pill) { pill.textContent = toneLabel(status.tone); pill.className = `workflow-status-pill rounded-full px-2 py-0.5 text-[10px] font-black ${stage.id === state.activeEditor ? 'bg-white/10 text-white' : statusPillClass(status.tone)}`; }
    if (tab) tab.dataset.tone = status.tone;
  });
  const current = progress[state.activeEditor];
  const label = document.querySelector('[data-workflow-current-label]');
  const reason = document.querySelector('[data-workflow-current-reason]');
  const bar = document.querySelector('[data-workflow-current-bar]');
  const card = document.querySelector('[data-workflow-current-status]');
  const nextReason = document.querySelector('[data-workflow-next-reason]');
  const index = WORKFLOW_STAGES.findIndex(stage => stage.id === state.activeEditor);
  const next = WORKFLOW_STAGES[index + 1];
  if (label) label.textContent = current.label;
  if (reason) reason.textContent = current.reason;
  if (bar) { bar.className = `h-full rounded-full ${progressBarClass(current.tone)}`; bar.style.width = `${current.percent}%`; }
  if (card) card.className = `min-w-[240px] rounded-2xl ${statusSurfaceClass(current.tone)} px-4 py-3`;
  if (nextReason && next) nextReason.textContent = progress[next.id].reason;
}

function setActiveEditor(editor) {
  if (!WORKFLOW_STAGE_IDS.has(editor) || state.activeEditor === editor) return;
  state.activeEditor = editor;
  try { globalThis.localStorage?.setItem(WORKFLOW_STORAGE_KEY, editor); } catch {}
  if (typeof history !== 'undefined') history.replaceState(null, '', `#workflow-${editor}`);
  rerenderShell();
}

function workspaceView() {
  return state.workspaceView[state.activeEditor] || (state.workspaceView[state.activeEditor] = { leftCollapsed: false, rightCollapsed: false });
}

function toggleWorkspaceColumn(side) {
  const view = workspaceView();
  if (side === 'left') view.leftCollapsed = !view.leftCollapsed;
  if (side === 'right') view.rightCollapsed = !view.rightCollapsed;
  rerenderShell();
}

function workspaceGridClass() {
  if (state.activeEditor === 'review') return 'xl:grid-cols-[minmax(0,1fr)_340px]';
  return 'xl:grid-cols-[360px_minmax(0,1fr)_340px]';
}

function stageShortName(id) { return ({ layout: '匯入', refine: '修圖', review: '檢查', package: '下載' })[id] || id; }
function toneLabel(tone) { return ({ complete: '完成', active: '進行中', warning: '警告', error: '錯誤', idle: '未開始' })[tone] || '未開始'; }
function activeAccentClass(accent) { return ({ emerald: 'text-emerald-300', rose: 'text-rose-300', sky: 'text-sky-300', amber: 'text-amber-300' })[accent] || 'text-white'; }
function accentClass(accent) { return ({ emerald: 'text-emerald-600', rose: 'text-rose-500', sky: 'text-sky-600', amber: 'text-amber-600' })[accent] || 'text-slate-500'; }
function accentBorderClass(accent) { return ({ emerald: 'border-emerald-300', rose: 'border-rose-300', sky: 'border-sky-300', amber: 'border-amber-300' })[accent] || 'border-white'; }
function statusSurfaceClass(tone) { return ({ complete: 'bg-emerald-50 text-emerald-800', active: 'bg-sky-50 text-sky-800', warning: 'bg-amber-50 text-amber-800', error: 'bg-rose-50 text-rose-800', idle: 'bg-slate-100 text-slate-600' })[tone] || 'bg-slate-100 text-slate-600'; }
function statusPillClass(tone) { return ({ complete: 'bg-emerald-50 text-emerald-700', active: 'bg-sky-50 text-sky-700', warning: 'bg-amber-50 text-amber-700', error: 'bg-rose-50 text-rose-700', idle: 'bg-slate-100 text-slate-500' })[tone] || 'bg-slate-100 text-slate-500'; }
function progressBarClass(tone) { return ({ complete: 'bg-emerald-500', active: 'bg-sky-500', warning: 'bg-amber-500', error: 'bg-rose-500', idle: 'bg-slate-300' })[tone] || 'bg-slate-300'; }
